# Rebuilds the progress table in README.md whenever a NOTES.md changes.
# Runs on GitHub's servers, so you never need to run the script yourself.
name: Update progress table

on:
  push:
    branches: [main]
    paths:
      - "**/NOTES.md"
      - "scripts/update_progress.py"
  workflow_dispatch:   # lets you run it by hand from the Actions tab

permissions:
  contents: write

jobs:
  update:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
      - run: python scripts/update_progress.py
      - name: Commit if the table changed
        run: |
          if git diff --quiet README.md; then
            echo "Progress table already up to date."
          else
            git config user.name "github-actions[bot]"
            git config user.email "41898283+github-actions[bot]@users.noreply.github.com"
            git add README.md
            git commit -m "Update progress table"
            git push
          fi
